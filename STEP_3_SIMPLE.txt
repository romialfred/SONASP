================================================================================
SQL TO EXECUTE (replace YOUR_USER_ID_HERE):
================================================================================

-- Option 1: Update existing profile
UPDATE user_profiles
SET 
  role = 'management',
  is_active = true
WHERE id = 'YOUR_USER_ID_HERE';

-- Option 2: If UPDATE returns 0 rows, create profile
INSERT INTO user_profiles (
  id,
  email,
  full_name,
  role,
  is_active,
  two_factor_enabled
)
SELECT 
  id,
  email,
  COALESCE(raw_user_meta_data->>'full_name', 'Admin User'),
  'management',
  true,
  false
FROM auth.users
WHERE id = 'YOUR_USER_ID_HERE'
ON CONFLICT (id) DO UPDATE
SET role = 'management', is_active = true;

-- Verify the result
SELECT 
  id,
  email,
  full_name,
  role,
  is_active
FROM user_profiles
WHERE id = 'YOUR_USER_ID_HERE';
