-- ═══════════════════════════════════════════════════════════════════════
-- DIAGNOSTIC COMPLET LICENSES
-- ═══════════════════════════════════════════════════════════════════════

-- 1. Vérifier tables
SELECT '1. TABLES' as check_type,
       COUNT(*) as count,
       CASE WHEN COUNT(*) >= 2 THEN '✅ OK' ELSE '❌ MANQUANT' END as status
FROM information_schema.tables
WHERE table_schema = 'public'
AND table_name IN ('licenses', 'license_requests');

-- 2. Vérifier vue
SELECT '2. VIEW' as check_type,
       COUNT(*) as count,
       CASE WHEN COUNT(*) = 1 THEN '✅ OK' ELSE '❌ MANQUANT' END as status
FROM pg_views
WHERE schemaname = 'public'
AND viewname = 'licenses_with_computed_fields';

-- 3. Compter licenses dans table
SELECT '3. LICENSES IN TABLE' as check_type,
       COUNT(*) as count,
       CASE WHEN COUNT(*) > 0 THEN '✅ OK' ELSE '❌ VIDE' END as status
FROM licenses;

-- 4. Vérifier RLS
SELECT '4. RLS ENABLED' as check_type,
       CASE WHEN rowsecurity THEN 1 ELSE 0 END as count,
       CASE WHEN rowsecurity THEN '✅ ACTIVÉ' ELSE '⚠️ DÉSACTIVÉ' END as status
FROM pg_tables
WHERE schemaname = 'public'
AND tablename = 'licenses';

-- 5. Vérifier votre rôle
SELECT '5. YOUR ROLE' as check_type,
       1 as count,
       COALESCE(role, '❌ PAS DE ROLE') as status
FROM user_profiles
WHERE id = auth.uid();
