    // ❌ Banques jamais sauvegardées!
  }

APRÈS (✅):
  handleSubmit() {
    let customerId;

    // 1. Créer/Mettre à jour customer
    if (isEditMode) {
      await supabase.from('customers').update(...);
      // Supprimer anciennes banques
      await supabase
        .from('customer_banks')
        .delete()
        .eq('customer_id', id);
      customerId = id;
    } else {
      const { data } = await supabase
        .from('customers')
        .insert(...)
        .select()
        .single();
      customerId = data.id;
    }

    // 2. ✅ Sauvegarder les banques!
    if (formData.banks.length > 0) {
      const banksToInsert = formData.banks.map(bank => ({
        customer_id: customerId,
        bank_name: bank.bankName,
        account_number: bank.accountNumber,
        swift_code: bank.swiftCode,
        iban: bank.iban,
        currency: bank.currency,
        country: bank.country,
        city: bank.city,
        is_primary: bank.isPrimary,
        is_active: bank.isActive,
      }));

      await supabase
        .from('customer_banks')
        .insert(banksToInsert);
    }
  }
