VERIFICATION QUERIES
================================================================================

After applying migrations, run these in SQL Editor:

-- Check license count (should return 10)
SELECT COUNT(*) FROM licenses;

-- View sample licenses
SELECT
  license_number,
  status,
  ROUND(authorized_qty_oz::numeric, 2) as auth_oz,
  ROUND(consumed_qty_oz::numeric, 2) as used_oz,
  ROUND(remaining_qty_oz::numeric, 2) as remaining_oz,
  days_to_expiry
FROM licenses
WHERE license_number LIKE 'LIC-2024-%'
ORDER BY created_at;

-- Check batches linked
SELECT
  l.license_number,
  COUNT(b.id) as batch_count
FROM licenses l
LEFT JOIN batches b ON b.license_id = l.id
WHERE l.license_number LIKE 'LIC-2024-%'
GROUP BY l.license_number;
