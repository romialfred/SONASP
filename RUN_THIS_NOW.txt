-- Simple check: Do license tables exist?
SELECT 
  COUNT(*) FILTER (WHERE table_name = 'licenses') as licenses_exists,
  COUNT(*) FILTER (WHERE table_name = 'license_requests') as requests_exists,
  COUNT(*) FILTER (WHERE table_name = 'license_events') as events_exists,
  COUNT(*) FILTER (WHERE table_name = 'license_quota_transactions') as transactions_exists,
  COUNT(*) FILTER (WHERE table_name = 'license_request_documents') as docs_exists,
  COUNT(*) FILTER (WHERE table_name = 'license_kpi_thresholds') as kpi_exists
FROM information_schema.tables
WHERE table_schema = 'public'
  AND table_name LIKE 'license%';

-- Check if view exists
SELECT 
  CASE 
    WHEN EXISTS (
      SELECT 1 FROM pg_views 
      WHERE schemaname = 'public' 
      AND viewname = 'licenses_with_computed_fields'
    ) THEN '✅ VIEW EXISTS'
    ELSE '❌ VIEW DOES NOT EXIST'
  END as view_status;


This WILL return rows showing:
  - How many license tables exist (should be 6)
  - If the view exists or not
